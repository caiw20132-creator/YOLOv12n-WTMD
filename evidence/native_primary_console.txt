>> def sha256_with_progress(path: Path, label: str) -> str:
>>     total = max(path.stat().st_size, 1)
>>     processed = 0
>>     next_report = 0.25
>>     digest = hashlib.sha256()
>>
>>     log(f"[SHA256] {label}")
>>     log(f"         path={path}")

>>             chunk = file.read(8 * 1024 * 1024)

>>                 break

>>             digest.update(chunk)
>>             processed += len(chunk)
